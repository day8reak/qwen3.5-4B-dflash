cmake_minimum_required(VERSION 3.16)
project(gdr_debug LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ASCENDCL_ROOT "" CACHE PATH "Active CANN toolkit root")
set(_hints "${ASCENDCL_ROOT}" "$ENV{ASCEND_HOME_PATH}" "$ENV{ASCEND_TOOLKIT_HOME}" "$ENV{CANN_HOME}")
find_path(ASCENDCL_INCLUDE_DIR acl/acl.h HINTS ${_hints} PATH_SUFFIXES include)
find_library(ASCENDCL_LIBRARY ascendcl HINTS ${_hints} PATH_SUFFIXES lib64 lib)
if(NOT ASCENDCL_INCLUDE_DIR OR NOT ASCENDCL_LIBRARY)
  message(FATAL_ERROR "Activate CANN or pass -DASCENDCL_ROOT=<toolkit-root>")
endif()
add_executable(gdr_debug_runner runner.cpp ../../framework/runtime/cpp/src/sha256.cpp)
target_include_directories(gdr_debug_runner PRIVATE "${ASCENDCL_INCLUDE_DIR}" ../../framework/runtime/cpp/include)
target_link_libraries(gdr_debug_runner PRIVATE "${ASCENDCL_LIBRARY}")
target_compile_options(gdr_debug_runner PRIVATE -Wall -Wextra -Wpedantic)
